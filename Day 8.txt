
1.Max sum of sub array using sliding window
implementation using array and fixed sliding window
public class Main {
    public static void main(String[] args) {
        int[] arr = {2, 1, 5, 1, 3, 2};
        int k = 3;
        int windowSum = 0;
        for (int i = 0; i < k; i++) {
            windowSum += arr[i];
        }
        int maxSum = windowSum;
        for (int i = k; i < arr.length; i++) {
            windowSum = windowSum
                        - arr[i - k]
                        + arr[i];
            maxSum = Math.max(maxSum, windowSum);
        }
        System.out.println("Maximum Sum = " + maxSum);
    }
}
2. First Negative Number in Every Window of Size K
Implementation using fixed sliding window and queue
Given an array and a window size k, print the first negative 
number in every window.
import java.util.*;
public class Main {
    public static void main(String[] args) {
        int[] arr = {12, -1, -7, 8, -15, 30, 16, 28};
        int k = 3;
        Queue<Integer> q = new LinkedList<>();
        int i = 0, j = 0;
        while (j < arr.length) {
            if (arr[j] < 0) {
                q.offer(arr[j]);
            }
            if (j - i + 1 < k) {
                j++;
            }
            else if (j - i + 1 == k) {
                if (q.isEmpty())
                    System.out.print("0 ");
                else
                    System.out.print(q.peek() + " ");
                if (!q.isEmpty() && arr[i] == q.peek()) {
                    q.poll();
                }
                i++;
                j++;
            }
        }
    }
}
3.Longest substring without non repeating chars
Implementation variable sliding window expand the window
when duplicate found shrink the window store the values in hashset
so no duplicates are allowed 

import java.util.*;
public class Main {
    public static void main(String[] args) {
        String s = "abcabcbb";
        int i = 0;
        int j = 0;
        int maxLen = 0;
        HashSet<Character> set = new HashSet<>();
        while (j < s.length()) {
            char ch = s.charAt(j);
            if (!set.contains(ch)) {
                set.add(ch);
                maxLen = Math.max(maxLen, j - i + 1);
                j++;
            }
            else {
                while (set.contains(ch)) {
                    set.remove(s.charAt(i));
                    i++;
                }
                set.add(ch);
                j++;
            }
        }
        System.out.println(maxLen);
    }
}
4. Maximum Number of Vowels in a Substring of Length K
public class Main {
    public static boolean isVowel(char ch) {
        return ch == 'a' || ch == 'e' || ch == 'i'
                || ch == 'o' || ch == 'u';
    }
    public static void main(String[] args) {
        String s = "abciiidef";
        int k = 3;
        int count = 0;
        int maxVowels = 0;
     for (int i = 0; i < k; i++) {
            if (isVowel(s.charAt(i))) {
                count++;
            }
        }
        maxVowels = count;
        for (int i = k; i < s.length(); i++) {
            if (isVowel(s.charAt(i - k))) {
                count--;
            }
           if (isVowel(s.charAt(i))) {
                count++;
            }
            maxVowels = Math.max(maxVowels, count);
        }
        System.out.println(maxVowels);
    }
}
5.Longest Subarray with Sum ≤ K
public class Main {
    public static void main(String[] args) {
        int[] arr = {1, 2, 1, 0, 1, 1, 0};
        int k = 4;
        int i = 0;
        int j = 0;
        int sum = 0;
        int maxLen = 0;
        while (j < arr.length) {
            sum += arr[j];
            while (sum > k) {
                sum -= arr[i];
                i++;
            }
            maxLen = Math.max(maxLen, j - i + 1);
            j++;
        }
        System.out.println("Longest Length = " + maxLen);
    }
}


import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int arr[] = new int[n];

        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }

        int k = sc.nextInt();
        k--;

        int low = 0, high = n - 1;

        while (low <= high) {
            int pivot = arr[high];
            int i = low;

            for (int j = low; j < high; j++) {
                if (arr[j] <= pivot) {
                    int temp = arr[i];
                    arr[i] = arr[j];
                    arr[j] = temp;
                    i++;
                }
            }

            int temp = arr[i];
            arr[i] = arr[high];
            arr[high] = temp;

            if (i == k) {
                System.out.println(arr[i]);
                break;
            } else if (i > k) {
                high = i - 1;
            } else {
                low = i + 1;
            }
        }
    }
