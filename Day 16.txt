insert method-use insertion sort
remove peek element
display 
public class Main {
     int[] heap = new int[20];   
    int size = 0;               
    public void insert(int value) {
        heap[size] = value;
        int current = size;
        while (current > 0) {
            int parent = (current - 1) / 2;
            if (heap[current] < heap[parent]) {
                int temp = heap[current];
                heap[current] = heap[parent];
                heap[parent] = temp;
                current = parent;
            } else {
                break;
            }
        }
        size++;
    }
    public void display() {
        for (int i = 0; i < size; i++) {
            System.out.print(heap[i] + " ");
        }
        System.out.println();
    }
    public int peek() {
        if (size == 0) {
            System.out.println("Heap is Empty");
            return -1;
        }
        return heap[0];
    }
    public int remove() {
        if (size == 0) {
            System.out.println("Heap is Empty");
            return -1;
        }
        int removed = heap[0];
        heap[0] = heap[size - 1];
        size--;
        int current = 0;
        while (true) {
            int left = 2 * current + 1;
            int right = 2 * current + 2;
            int smallest = current;
            if (left < size && heap[left] < heap[smallest]) {
                smallest = left;
            }
            if (right < size && heap[right] < heap[smallest]) {
                smallest = right;
            }

            if (smallest != current) {
                int temp = heap[current];
                heap[current] = heap[smallest];
                heap[smallest] = temp;
                current = smallest;
            } else {
                break;
            }
        }
        return removed;
    }

public static void main(String[] args) {
        Main obj = new Main();
        obj.insert(45);
        obj.insert(12);
        obj.insert(78);
        obj.insert(3);
        obj.insert(56);
        obj.insert(90);
        obj.insert(1);
        obj.insert(67);
        obj.insert(23);
        obj.insert(8);
        System.out.println("Heap Elements");
        obj.display();
        System.out.println("Peek Element = " + obj.peek());
        System.out.println("Deleted Element = " + obj.remove());
        System.out.println("Heap After Deletion");
        obj.display();
    }
}

2. Find the kth smallest element
Sort the array
get k from user 
find the kth min element
eg:-ar={12,3,5,7,19}

output :7

import java.util.*;
public class Main {
    public static void sortArray(int arr[]) {
        int n = arr.length;
        for (int i = 0; i < n - 1; i++) {
            for (int j = 0; j < n - i - 1; j++) {
                if (arr[j] > arr[j + 1]) {
                    int temp = arr[j];
                    arr[j] = arr[j + 1];
                    arr[j + 1] = temp;
                }
            }
        }
    }
    public static int kthSmallest(int arr[], int k) {
        sortArray(arr);
        return arr[k - 1];
    }
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

            System.out.println("Invalid value of k");
        } else {

             System.out.print(arr[i] + " ");
            }

        }  
    }

4. Merge k sorted array  create a min heap
a={1,4,5}
b={2,6,7}

Store the first element of each array in a heap
remove the min element
repeat the process in the same array
TC:0(n log k)
public class Main {
 static class Node {
        int value;      
        int row;        
        int col;        
        Node(int value, int row, int col) {
            this.value = value;
            this.row = row;
            this.col = col;
        }
    }
    static Node[] heap = new Node[10];
    static int size = 0;
    static void insert(Node node) {
        heap[size] = node;
        int current = size;
        while (current > 0) {
            int parent = (current - 1) / 2;
            if (heap[current].value < heap[parent].value) {
                Node temp = heap[current];
                heap[current] = heap[parent];
                heap[parent] = temp;
                current = parent;

            } else {
                break;
            }
        }

        size++;
    }

    static Node remove() {
        Node removed = heap[0];
        heap[0] = heap[size - 1];
        size--;
        int current = 0;
        while (true) {
            int left = 2 * current + 1;
            int right = 2 * current + 2;
            int smallest = current;
            if (left < size &&
                    heap[left].value < heap[smallest].value)
                smallest = left;
            if (right < size &&
                    heap[right].value < heap[smallest].value)
                smallest = right;
            if (smallest != current) {
                Node temp = heap[current];
                heap[current] = heap[smallest];
                heap[smallest] = temp;
                current = smallest;

            } else {
                break;
            }
        }

        return removed;
    }

                {1, 3, 5},

        };

                int temp = heap[current];

    public int remove() {
        int removed = heap[0];

        size--;
        int current = 0;
        while (true) {
            int left = 2 * current + 1;

    public static void main(String[] args) {
