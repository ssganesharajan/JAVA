System.out.println("Enter no of elements:");        n=sc.nextInt();        int ar[]=new int[n];        System.out.println("Enter array elements:");        for(i=0;i<n;i++){            ar[i]=sc.nextInt();            }            int largest=ar[0];            int secondlargest=ar[0];        for(i=1;i<n;i++){            if(ar[i]>largest){                secondlargest=largest;                largest=ar[i];            }            else if(ar[i]>secondlargest && ar[i]!=largest){                secondlargest=ar[i];            }        }        System.out.println("Second largest: "+secondlargest);    }}---Stack, Queue can be implemented using array and list but array not recommended because of continuous memory allocationStack - Push(), Pop(), Peek()Queue -
