1. Move all the zero's to the end 

public class Main {
    public static void main(String[] args) {
        int[] arr = {0, 1, 0, 3, 12};
        int j = 0;
        for (int i = 0; i < arr.length; i++) {
            if (arr[i] != 0) {
                int temp = arr[i];
                arr[i] = arr[j];
                arr[j] = temp;
                j++;
            }
        }
        for (int x : arr) {
            System.out.print(x + " ");
        }
    }
}
2.Kanden's algorithm
public class Main {
    public static void main(String[] args) {
        int[] arr = {-2, 1, -3, 4, -1, 2, 1, -5, 4};
        int currentSum = 0;
        int maxSum = Integer.MIN_VALUE;
        for (int i = 0; i < arr.length; i++) {
            currentSum = currentSum + arr[i];
            if (currentSum > maxSum) {
                maxSum = currentSum;
            }
            if (currentSum < 0) {
                currentSum = 0;
            }
        }
        System.out.println("Maximum Subarray Sum = " + maxSum);
    }
}
3.Equilibrium Index
public class Main {
    public static void main(String[] args) {
        int[] arr = {1, 3, 5, 2, 2};
        int totalSum = 0;
        // Calculate total sum
        for (int num : arr) {
            totalSum += num;
        }
        int leftSum = 0;
        for (int i = 0; i < arr.length; i++) {
            totalSum = totalSum - arr[i]; // right sum
            if (leftSum == totalSum) {
                System.out.println("Equilibrium Index = " + i);
                return;
            }
            leftSum = leftSum + arr[i];
        }
        System.out.println("No Equilibrium Index Found");
    }
}
4.Leaders in array
public class Main {
    public static void main(String[] args) {
        int[] arr = {16, 17, 4, 3, 5, 2};
        int maxRight = arr[arr.length - 1];
        System.out.print("Leaders: ");
        System.out.print(maxRight + " ");
        for (int i = arr.length - 2; i >= 0; i--) {
            if (arr[i] > maxRight) {
                maxRight = arr[i];
                System.out.print(maxRight + " ");
            }
        }
    }
}
5.Majority elements
Using Brute Force TC : O(n^2)
public class Main {
    public static void main(String[] args) {
        int[] arr = {2, 2, 1, 2, 3, 2, 2};
        int n = arr.length;
        for (int i = 0; i < n; i++) {
            int count = 0;
            for (int j = 0; j < n; j++) {
                if (arr[i] == arr[j]) {
                    count++;
                }
            }
            if (count > n / 2) {
                System.out.println("Majority Element = " + arr[i]);
                return;
            }
        }
        System.out.println("No Majority Element");
    }
}
M2: Using Hash map
import java.util.HashMap;
public class Main {
    public static void main(String[] args) {
        int[] arr = {2, 2, 1, 2, 3, 2, 2};
        int n = arr.length;
        HashMap<Integer, Integer> map = new HashMap<>();
        for (int num : arr) {
            map.put(num, map.getOrDefault(num, 0) + 1);
            if (map.get(num) > n / 2) {
                System.out.println("Majority Element = " + num);
                return;
            }
        }
        System.out.println("No Majority Element");
    }
}
7. Pair with target sum
public class Main {
    public static void main(String[] args) {
        int[] arr = {1, 2, 3, 4, 6};
        int target = 6;
        int left = 0;
        int right = arr.length - 1;
        boolean found = false;
        while (left < right) {
            int sum = arr[left] + arr[right];
            if (sum == target) {
                System.out.println("Pair Found: "
                        + arr[left] + " + "
                        + arr[right] + " = "
                        + target);
                found = true;
                break;
            }
            else if (sum < target) {
                left++;
            }
            else {
                right--;
            }
        }
        if (!found) {
            System.out.println("No Pair Found");
        }
    }
}
