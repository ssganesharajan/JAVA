Heap does not display elements in sorted order, why?
	Min heap tree - Root node should be smaller than left and right nodes.
eg: 1,8,23,56,7,12,45 instead of 1,7,8,12,23,45,56

this is not sorted because this is not array to sort, we can find only min and max.

implementation of minheap using priority queue ascending max comparator not working for descending
For maxheap priority queue we have to trigger it
eg: 45,12,78,3,56,90 for maxheap 90,45,3,12,78,56

Implementation of max heap practice
1. build a max heap from an array
2. insert element into max heap and display
3. print the top most element without removing
4. remove the top most element
5. do heap sort with using collections ascending order

import java.util.*;

public class Main {
    ArrayList<Integer> heap;

    public Main() {
        heap = new ArrayList<>();
    }

    // Build Max Heap from an array
    public void buildHeap(int arr[]) {
        for (int i = 0; i < arr.length; i++) {
            insert(arr[i]);
        }
    }

    // Insert into Max Heap
    public void insert(int value) {
        heap.add(value);
        int current = heap.size() - 1;

        while (current > 0) {
            int parent = (current - 1) / 2;

            if (heap.get(parent) < heap.get(current)) {
                int temp = heap.get(parent);
                heap.set(parent, heap.get(current));
                heap.set(current, temp);
                current = parent;
            } else {
                break;
            }
        }
    }

    // Peek (Top Element)
    public int peek() {
        if (heap.isEmpty())
            return -1;
        return heap.get(0);
    }

    // Remove Top Element
    public int remove() {
        if (heap.isEmpty())
            return -1;

        int max = heap.get(0);

        heap.set(0, heap.get(heap.size() - 1));
        heap.remove(heap.size() - 1);

        if (!heap.isEmpty())
            heapifyDown(0);

        return max;
    }

    // Heapify Down
    private void heapifyDown(int current) {
        while (true) {
            int left = 2 * current + 1;
            int right = 2 * current + 2;
            int largest = current;

            if (left < heap.size() && heap.get(left) > heap.get(largest))
                largest = left;

            if (right < heap.size() && heap.get(right) > heap.get(largest))
                largest = right;

            if (largest == current)
                break;

            int temp = heap.get(current);
            heap.set(current, heap.get(largest));
            heap.set(largest, temp);

            current = largest;
        }
    }

    // Display Heap
    public void display() {
        System.out.println(heap);
    }

    // Heap Sort (Ascending)
    public void heapSortAscending() {
        ArrayList<Integer> temp = new ArrayList<>(heap);
        ArrayList<Integer> sorted = new ArrayList<>();

        while (!heap.isEmpty()) {
            sorted.add(remove());
        }

        Collections.reverse(sorted);

        System.out.println("Heap Sort (Ascending): " + sorted);

        heap = temp;
    }

    public static void main(String[] args) {

        Main h = new Main();

        int arr[] = {45, 12, 78, 3, 56, 90};

        // 1. Build Max Heap
        h.buildHeap(arr);
        System.out.println("Max Heap:");
        h.display();

        // 2. Insert Element
        h.insert(100);
        System.out.println("After inserting 100:");
        h.display();

        // 3. Peek Top Element
        System.out.println("Top Element: " + h.peek());

        // 4. Remove Top Element
        System.out.println("Removed Element: " + h.remove());
        System.out.println("After Removal:");
        h.display();

        // 5. Heap Sort in Ascending Order
	h.heapSortAscending();
    }
}

import java.util.*;
public class Main{
    ArrayList<ArrayList<Integer>> graph;
    public Main(int vertices){
        graph = new ArrayList<>();
        for(int i=0;i<vertices;i++){
            graph.add(new ArrayList<>());
        }
    }
    
    public void addEdge(int source, int destination){
        graph.get(source).add(destination);
        graph.get(destination).add(source); //Unidirectional graph
    }
    
    public void display(){
        for(int i=0;i<graph.size();i++){
            System.out.print(i+" -> ");
            for(int neighbour : graph.get(i)){
                System.out.print(neighbour+" ");
            }
            System.out.println();
        }
    }
    
    public static void main(String[] args){
        Main m= new Main(4);
        m.addEdge(0,1);
        m.addEdge(0,2);
        m.addEdge(1,2);
        m.addEdge(1,3);
        m.display();
    }
}

import java.util.*;
public class Main{
    ArrayList<ArrayList<Integer>> graph;
    public Main(int vertices){
        graph = new ArrayList<>();
        for(int i=0;i<vertices;i++){
            graph.add(new ArrayList<>());
        }
    }
    
    public void addEdge(int source, int destination){
        graph.get(source).add(destination);
        graph.get(destination).add(source); //Unidirectional graph
    }
    
    public void bfs(int start){
        boolean visited[] = new boolean[graph.size()];
        Queue<Integer> queue = new LinkedList<>();
        visited[start]= true;
        queue.offer(start);
        while(!queue.isEmpty()){
            int current = queue.poll();
            System.out.print(current+" ");
            for (int neighbour: graph.get(current)){
                if(!visited[neighbour]){
                    visited[neighbour]=true;
                    queue.offer(neighbour);
                }
            }
        }
    }
    
    public static void main(String[] args){
        Main m= new Main(4);
        m.addEdge(0,1);
        m.addEdge(0,2);
        m.addEdge(1,2);
        m.addEdge(1,3);
        m.bfs(0);
    }
}

import java.util.*;
public class Main{
    ArrayList<ArrayList<Integer>> graph;
    public Main(int vertices){
        graph = new ArrayList<>();
        for(int i=0;i<vertices;i++){
            graph.add(new ArrayList<>());
        }
    }
    
    public void addEdge(int source, int destination){
        graph.get(source).add(destination);
        graph.get(destination).add(source); //Unidirectional graph
    }
    
    public void dfs(int start){
        boolean visited[] = new boolean[graph.size()];
        Stack<Integer> stack = new Stack<>();
        stack.push(start);
        while(!stack.isEmpty()){
            int current = stack.pop();
            if(!visited[current]){
                visited[current] = true;
                System.out.print(current + " ");
                for(int i=graph.get(current).size()-1;i>=0;i--){
                    int neighbour=graph.get(current).get(i);
                    if(!visited[neighbour]){
                        stack.push(neighbour);
                    }
                }
            }
        }
    }
    
    public static void main(String[] args){
        Main m= new Main(4);
        m.addEdge(0,1);
        m.addEdge(0,2);
        m.addEdge(1,2);
        m.addEdge(1,3);
        m.dfs(0);
    }
}
