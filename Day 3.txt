Encapsulation:

import java.util.*;

class Student {
	private String dept;
	private String number;
	private int dob;

	public void setDept(String dept) {
		this.dept = dept;
	}
	public void setMobileNumber(String number) {
		this.number = number;
	}
	public void setDob(int dob) {
		this.dob = dob;
	}
	public String getNumber() {
		return number;
	}
	public int getDob() {
		return dob;
	}
	public String getDept() {
		return dept;
	}
}
public class Main {
	public static void main(String[] args) {
		Student s = new Student();
		Scanner sc = new Scanner(System.in);

		System.out.println("Enter Your DEPT:- ");
		s.setDept(sc.next());
		System.out.println("Enter your Mobile number:- ");
		s.setMobileNumber(sc.next());
		System.out.println("Enter your D.O.B:- ");
		s.setDob(sc.nextInt());
		System.out.println("Mobile Number: " + s.getNumber());
		System.out.println("Date of Birth: " + s.getDob());
		System.out.println("Department: " + s.getDept());
	}
}



output:-

Enter Your DEPT:- 
CSE
Enter your Mobile number:- 
8140707090
Enter your D.O.B:- 
4/8/2006
