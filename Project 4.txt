Library Management System
s1. User management, Register, Update member details, Delete members
s2. Book management, add new book, update book info, delete book, search book by title, author, ISBN or category
s3, Issue and return book, issue books to members, return books, calculate overdue find automatically
s4. Fine management, calculate fine late returns record fine.
s5. Reports, available books, issued book, overdue book, borrowing item

import java.util.*;
import java.time.LocalDate;
import java.time.temporal.ChronoUnit;

// ====================== MODELS ======================
class Book {
    String isbn, title, author, category;
    boolean available;
    
    public Book(String isbn, String title, String author, String category) {
        this.isbn = isbn; 
        this.title = title; 
        this.author = author;
        this.category = category; 
        this.available = true;
    }
    
    public String toString() {
        return String.format("[%s] %s by %s (%s) - %s", 
                isbn, title, author, category, available ? "Available" : "Issued");
    }
}

class Member {
    String memberId, name, email;
    List<String> borrowed = new ArrayList<>();
    
    public Member(String id, String name, String email) {
        this.memberId = id; 
        this.name = name; 
        this.email = email;
    }
    
    public String toString() { 
        return memberId + " - " + name; 
    }
}

// ====================== INTERFACES ======================
interface IUserManagement {
    void addMember(Member m);
    void updateMember(String id, String name, String email);
    void deleteMember(String id);
    Member findMember(String id);
    List<Member> getAllMembers();
}

interface IBookManagement {
    void addBook(Book b);
    void updateBook(String isbn, String title, String author, String category);
    void deleteBook(String isbn);
    Book findBook(String isbn);
    List<Book> searchBooks(String keyword);  // Search by title, ISBN, author
    List<Book> getAvailableBooks();
}

public class Main {
    public static void main(String[] args) {
        LibrarySystem lib = new LibrarySystem();
        Scanner sc = new Scanner(System.in);

        lib.addBook(new Book("001", "Java Programming", "James Gosling", "Programming"));
        lib.addBook(new Book("002", "Clean Code", "Robert Martin", "Programming"));

    }

// ====================== CORE IMPLEMENTATION ======================
class LibrarySystem implements IUserManagement, IBookManagement, ILendingService, IFineService {
    private Map<String, Member> members = new HashMap<>();
    private Map<String, Book> books = new HashMap<>();
    private Map<String, LocalDate> dueDates = new HashMap<>();

    // Book Management
    public void addBook(Book b) { books.put(b.isbn, b); }
    
    public void updateBook(String isbn, String title, String author, String category) {
        Book b = books.get(isbn);
        if (b != null) {
            b.title = title;
            b.author = author;
            b.category = category;
            System.out.println("Book updated!");
        } else {
            System.out.println("Book not found!");
        }
    }
    
    public void deleteBook(String isbn) {
        if (books.remove(isbn) != null) System.out.println("Book deleted!");
        else System.out.println("Book not found!");
    }
    
    public Book findBook(String isbn) { return books.get(isbn); }
    
    public List<Book> searchBooks(String keyword) {
        List<Book> result = new ArrayList<>();
        String k = keyword.toLowerCase();
        for (Book b : books.values()) {
            if (b.title.toLowerCase().contains(k) || 
                b.author.toLowerCase().contains(k) || 
                b.isbn.toLowerCase().contains(k)) {
                result.add(b);
            }
        }
        return result;
    }
    
    public List<Book> getAvailableBooks() {
        List<Book> list = new ArrayList<>();
        for (Book b : books.values()) if (b.available) list.add(b);
        return list;
    }

    // User Management
    public void addMember(Member m) { members.put(m.memberId, m); }
    
    public void updateMember(String id, String name, String email) {
        Member m = members.get(id);
        if (m != null) {
            m.name = name;
            m.email = email;
            System.out.println("Member updated!");
        } else {
            System.out.println("Member not found!");
        }
    }
    
    public void deleteMember(String id) {
        if (members.remove(id) != null) System.out.println("Member deleted!");
        else System.out.println("Member not found!");
    }
    
    public Member findMember(String id) { return members.get(id); }
    public List<Member> getAllMembers() { return new ArrayList<>(members.values()); }

    // Lending & Fine
    public boolean issueBook(String memberId, String isbn) {
        Member m = findMember(memberId);
        Book b = findBook(isbn);
        if (m == null || b == null || !b.available) return false;
        b.available = false;
        m.borrowed.add(isbn);
        dueDates.put(memberId + "_" + isbn, LocalDate.now().plusDays(14));
        return true;
    }

    public boolean returnBook(String memberId, String isbn) {
        Member m = findMember(memberId);
        Book b = findBook(isbn);
        if (m == null || b == null) return false;
        b.available = true;
        m.borrowed.remove(isbn);
        dueDates.remove(memberId + "_" + isbn);
        return true;
    }

    public double calculateFine(String memberId, String isbn) {
        LocalDate due = dueDates.get(memberId + "_" + isbn);
        if (due == null) return 0;
        long days = ChronoUnit.DAYS.between(due, LocalDate.now());
        return days > 0 ? days * 5.0 : 0;
    }
}

// Simple interface for compilation
interface ILendingService {
    boolean issueBook(String memberId, String isbn);
    boolean returnBook(String memberId, String isbn);
}

interface IFineService {
    double calculateFine(String memberId, String isbn);
}
