class Main extends Thread{
    @Override
    public void run(){
        System.out.println("Thread is running...");
        System.out.println("Current thread: "+Thread.currentThread().getName());
        for(int i=1;i<=5;i++){
            System.out.println("Count= "+i);
            try{
                Thread.sleep(5000);
            }
            catch(InterruptedException e){
                System.out.println("Thread interrupted");
            }
        }
        System.out.println("Thread execution completed.");
    }
    
    public static void main(String[] args){
        System.out.println("Main thread started");
        Main t = new Main();
        t.start();
        System.out.println("Main thread continues...");
        System.out.println("Main thread Name: " + Thread.currentThread().getName());
        System.out.println("Main thread finished");
    }
}

//Greedy
import java.util.*;

class Item {
    int profit, weight;

    Item(int profit, int weight) {
        this.profit = profit;
        this.weight = weight;
    }
}

public class Main {
    public static void main(String[] args) {

        int capacity = 50;

        Item[] items = {
            new Item(60, 10),
            new Item(100, 20),
            new Item(120, 30)
        };

        // Sort by profit/weight ratio in descending order
        Arrays.sort(items, (a, b) -> {
            double r1 = (double) a.profit / a.weight;
            double r2 = (double) b.profit / b.weight;
            return Double.compare(r2, r1);
        });

        double totalProfit = 0.0;

        for (Item item : items) {

            if (capacity >= item.weight) {
                capacity -= item.weight;
                totalProfit += item.profit;

                System.out.println("Taken full item -> Profit: "
                        + item.profit + ", Weight: " + item.weight);
            } else {
                double fraction = (double) capacity / item.weight;
                totalProfit += item.profit * fraction;

                System.out.println("Taken "
                        + (fraction * 100) + "% of item -> Profit: "
                        + (item.profit * fraction));
                break;
            }
        }

        System.out.println("Maximum Profit = " + totalProfit);
    }
}

//Greedy
import java.util.*;

class Activity{
    int start, end;
    Activity(int start, int end){
        this.start= start;
        this.end = end;
    }
}

public class Main{
    public static void main(String[] args){
        Activity[] activities ={
            new Activity(1,3),
            new Activity(2,4),
            new Activity(3,5),
            new Activity(0,6),
            new Activity(5,7),
            new Activity(8,9)
        };
        
        //sort it by end time
        Arrays.sort(activities,(a,b) -> a.end-b.end);
        System.out.println("Selected Activities: ");
        
        //select first Activity
        int lastEnd = activities[0].end;
        System.out.println("("+activities[0].start+","+activities[0].end+")");
        
        //check remaining activities
        for (int i=1;i<activities.length;i++){
            if(activities[i].start>=lastEnd){
                System.out.println("("+activities[i].start+","+activities[i].end+")");
                lastEnd = activities[i].end;
            }
        }
    }
}

//Greedy
import java.util.*;

class Job {
    String id;
    int deadline;
    int profit;

    Job(String id, int deadline, int profit) {
        this.id = id;
        this.deadline = deadline;
        this.profit = profit;
    }
}

public class Main {
    public static void main(String[] args) {

        Job[] jobs = {
            new Job("J1", 2, 100),
            new Job("J2", 1, 19),
            new Job("J3", 2, 27),
            new Job("J4", 1, 25),
            new Job("J5", 3, 15)
        };

        // Sort jobs by profit (descending)
        Arrays.sort(jobs, (a, b) -> b.profit - a.profit);

        // Find maximum deadline
        int maxDeadline = 0;
        for (Job job : jobs) {
            maxDeadline = Math.max(maxDeadline, job.deadline);
        }

        // Create slots
        String[] slots = new String[maxDeadline];
        int totalProfit = 0;

        // Schedule jobs
        for (Job job : jobs) {
            for (int j = job.deadline - 1; j >= 0; j--) {
                if (slots[j] == null) {
                    slots[j] = job.id;
                    totalProfit += job.profit;
                    break;
                }
            }
        }

        System.out.println("Selected Jobs:");
        for (String s : slots) {
            if (s != null) {
                System.out.print(s + " ");
            }
        }

        System.out.println("\nTotal Profit: " + totalProfit);
    }
}

//Greedy
import java.util.*;

public class Main{
    public static void main(String[] args){
        int greed[]={1,2,3};
        int cookies[]={1,1};
        
        Arrays.sort(greed);
        Arrays.sort(cookies);
        
        int child = 0;
        int cookie = 0;
        int satisfied=0;
        
        while(child<greed.length && cookie<cookies.length){
            if(cookies[cookie]>=greed[child]){
                satisfied++;
                child++;
                cookie++;
            }
            else{
                cookie++;
            }
        }
        System.out.println("satisfied child:"+satisfied);
    }
}
