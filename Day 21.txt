class Employee {
    private String name;
    private double salary;

    public Employee(String name, double salary) {
        this.name = name;
        this.salary = salary;
    }

    public String getName() {
        return name;
    }

    public double getSalary() {
        return salary;
    }
}

// Abstraction (High-level depends on this interface)
interface SalaryService {
    double calculateIncentive(Employee emp);
    double calculateNetSalary(Employee emp);
}

// Low-level module
class EmployeeSalaryCalculator implements SalaryService {

    @Override
    public double calculateIncentive(Employee emp) {
        return emp.getSalary() * 0.05;
    }

    @Override
    public double calculateNetSalary(Employee emp) {
        return emp.getSalary() + calculateIncentive(emp);
    }
}

// Another low-level module
class ManagerSalaryCalculator implements SalaryService {

    @Override
    public double calculateIncentive(Employee emp) {
        return emp.getSalary() * 0.10;
    }

    @Override
    public double calculateNetSalary(Employee emp) {
        return emp.getSalary() + calculateIncentive(emp);
    }
}

// Repository (SRP)
class EmployeeRepository {

    public void save(Employee emp) {
        System.out.println("Saving " + emp.getName() + " to database.");
    }
}

// High-level module depends on abstraction
class Report {

    private SalaryService salaryService;

    public Report(SalaryService salaryService) {
        this.salaryService = salaryService;
    }

    public void generateReport(Employee emp) {
        System.out.println("\n------ Salary Report ------");
        System.out.println("Employee   : " + emp.getName());
        System.out.println("Salary     : " + emp.getSalary());
        System.out.println("Incentive  : " + salaryService.calculateIncentive(emp));
        System.out.println("Net Salary : " + salaryService.calculateNetSalary(emp));
    }
}

public class Main {

    public static void main(String[] args) {

        Employee emp = new Employee("Alice", 50000);

        EmployeeRepository repository = new EmployeeRepository();
        repository.save(emp);

        // Runtime polymorphism (Dependency Injection)
        SalaryService service = new EmployeeSalaryCalculator();
        Report report = new Report(service);
        report.generateReport(emp);

        service = new ManagerSalaryCalculator();
        report = new Report(service);
        report.generateReport(emp);
    }
}

import java.util.Scanner;
class Employee {
    private int empId;
    private String empName;
    private double basicSalary;
    private double hra;
    Employee(int empId, String empName, double basicSalary, double hra) {

        this.basicSalary = basicSalary;
        this.hra = hra;

    public int getEmpId() {
        return empId;
    }

        return empName;
    }
    public double getBasicSalary() {
        return basicSalary;
    }
    public double getHra() {
        return hra;
    }
}
class SalaryCalculator {
    private double incentive;
    private double netSalary;
    public void calculateSalary(Employee emp) {
        incentive = emp.getBasicSalary() * 0.05;
        netSalary = emp.getBasicSalary()
                + emp.getHra()
                + incentive;
    }
    public double getIncentive() {
        return incentive;
    }
    public double getNetSalary() {
        return netSalary;
    }
}

class ReportGenerator {
    public void generateReport(Employee emp, SalaryCalculator salary) {
        System.out.println("\n========== EMPLOYEE REPORT ==========");
        System.out.println("Employee ID      : " + emp.getEmpId());
        System.out.println("Employee Name    : " + emp.getEmpName());
        System.out.println("Basic Salary     : " + emp.getBasicSalary());
        System.out.println("HRA              : " + emp.getHra());
        System.out.println("Incentive (5%)   : " + salary.getIncentive());
        System.out.println("Net Salary       : " + salary.getNetSalary());
        System.out.println("=====================================");
    }
}

Student management system with SOLID principles in java
s1. studententity with firlds,getter,setter, constructor
s2. students repository add(),display(),search(),delete()
s3. student service update marks, calculate grade, calculate percentage
s4. Report generate student report
s5. Main menu driven dynamic input Add, Display, Search, Update, Delete, Report

import java.util.ArrayList;
import java.util.Scanner;

class Student {
    private int id;
    private String name;
    private int marks1;
    private int marks2;
    private int marks3;

    public Student(int id, String name, int marks1, int marks2, int marks3) {
        this.id = id;
        this.name = name;
        this.marks1 = marks1;
        this.marks2 = marks2;
        this.marks3 = marks3;
    }

    public int getId() {
        return id;
    }

    public String getName() {
        return name;
    }

    public int getMarks1() {
        return marks1;
    }

    public int getMarks2() {
        return marks2;
    }

    public int getMarks3() {
        return marks3;
    }

    public void setMarks1(int marks1) {
        this.marks1 = marks1;
    }

    public void setMarks2(int marks2) {
        this.marks2 = marks2;
    }

    public void setMarks3(int marks3) {
        this.marks3 = marks3;
    }
}

class StudentRepository {

    private ArrayList<Student> students = new ArrayList<>();

    public void add(Student student) {
        students.add(student);
    }

    public void display() {
        if (students.isEmpty()) {
            System.out.println("No students available.");
            return;
        }

        for (Student s : students) {
            System.out.println(s.getId() + " " + s.getName());
        }
    }

    public Student search(int id) {
        for (Student s : students) {
            if (s.getId() == id)
                return s;
        }
        return null;
    }

    public boolean delete(int id) {
        Student s = search(id);

        if (s != null) {
            students.remove(s);
            return true;
        }

        return false;
    }
}

class StudentService {

    public void updateMarks(Student student, int m1, int m2, int m3) {
        student.setMarks1(m1);
        student.setMarks2(m2);
        student.setMarks3(m3);
    }

    public double calculatePercentage(Student student) {
        return (student.getMarks1() + student.getMarks2() + student.getMarks3()) / 3.0;
    }

    public String calculateGrade(Student student) {

        double per = calculatePercentage(student);

        if (per >= 90)
            return "A";

        if (per >= 75)
            return "B";

        if (per >= 60)
            return "C";

        if (per >= 40)
            return "D";

        return "F";
    }
}

class ReportGenerator {

    private StudentService service = new StudentService();

    public void generateStudentReport(Student student) {

        if (student == null) {
            System.out.println("Student not found.");
            return;
        }

        System.out.println("\n----- Student Report -----");
        System.out.println("ID : " + student.getId());
        System.out.println("Name : " + student.getName());
        System.out.println("Marks1 : " + student.getMarks1());
        System.out.println("Marks2 : " + student.getMarks2());
        System.out.println("Marks3 : " + student.getMarks3());
        System.out.println("Percentage : " + service.calculatePercentage(student));
        System.out.println("Grade : " + service.calculateGrade(student));
    }
}

public class Main {

    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        StudentRepository repo = new StudentRepository();
        StudentService service = new StudentService();
        ReportGenerator report = new ReportGenerator();

        while (true) {

            System.out.println("\n===== Student Management System =====");
            System.out.println("1. Add Student");
            System.out.println("2. Display Students");
            System.out.println("3. Search Student");
            System.out.println("4. Update Marks");
            System.out.println("5. Delete Student");
            System.out.println("6. Generate Report");
            System.out.println("7. Exit");

            System.out.print("Enter Choice: ");
            int choice = sc.nextInt();

            switch (choice) {

                case 1:

                    System.out.print("Enter ID: ");
                    int id = sc.nextInt();
                    sc.nextLine();

                    System.out.print("Enter Name: ");
                    String name = sc.nextLine();

                    System.out.print("Enter Marks1: ");
                    int m1 = sc.nextInt();

                    System.out.print("Enter Marks2: ");
                    int m2 = sc.nextInt();

                    System.out.print("Enter Marks3: ");
                    int m3 = sc.nextInt();

                    repo.add(new Student(id, name, m1, m2, m3));
                    System.out.println("Student Added.");
                    break;

                case 2:
                    repo.display();
                    break;

                case 3:

                    System.out.print("Enter ID: ");
                    id = sc.nextInt();

                    Student s = repo.search(id);

                    if (s != null)
                        System.out.println("Found : " + s.getName());
                    else
                        System.out.println("Student Not Found");

                    break;

                case 4:

                    System.out.print("Enter ID: ");
                    id = sc.nextInt();

                    s = repo.search(id);

                    if (s != null) {

                        System.out.print("Enter New Marks1: ");
                        m1 = sc.nextInt();

                        System.out.print("Enter New Marks2: ");
                        m2 = sc.nextInt();

                        System.out.print("Enter New Marks3: ");
                        m3 = sc.nextInt();

                        service.updateMarks(s, m1, m2, m3);

                        System.out.println("Marks Updated.");
                    } else {
                        System.out.println("Student Not Found");
                    }

                    break;

                case 5:

                    System.out.print("Enter ID: ");
                    id = sc.nextInt();

                    if (repo.delete(id))
                        System.out.println("Deleted Successfully");
                    else
                        System.out.println("Student Not Found");

                    break;

                case 6:

                    System.out.print("Enter ID: ");
                    id = sc.nextInt();

                    s = repo.search(id);
                    report.generateStudentReport(s);

                    break;

                case 7:
                    System.out.println("Thank You");
                    System.exit(0);

                default:
                    System.out.println("Invalid Choice");
            }
        }
    }
}
